function buscarMedicamento() { const texto = document .getElementById("pesquisa") .value .toLowerCase() .trim(); const resultados = document.getElementById("resultados"); const ficha = document.getElementById("ficha"); ficha.style.display = "none"; if (texto === "") { resultados.innerHTML = ""; return; } const encontrados = medicamentos.filter(medicamento => medicamento.nome.toLowerCase().includes(texto) || medicamento.principio.toLowerCase().includes(texto) || medicamento.classe.toLowerCase().includes(texto) || medicamento.especies.some(especie => especie.toLowerCase().includes(texto) ) ); if (encontrados.length === 0) { resultados.innerHTML = `
😕 Nenhum medicamento encontrado.
`; return; } resultados.innerHTML = encontrados.map(medicamento => `

💊 ${medicamento.nome}

Classe: ${medicamento.classe}

Princípio ativo: ${medicamento.principio}

`).join(""); } function abrirFicha(nome) { const medicamento = medicamentos.find( item => item.nome === nome ); if (!medicamento) return; const ficha = document.getElementById("ficha"); const resultados = document.getElementById("resultados"); resultados.innerHTML = ""; ficha.style.display = "block"; ficha.innerHTML = `

💊 ${medicamento.nome}

${medicamento.classe}

🧪 Princípio ativo

${medicamento.principio}

⚙️ Mecanismo de ação

${medicamento.mecanismo}

🐾 Espécies

${medicamento.especies.join(" • ")}

🎯 Indicações

${medicamento.indicacoes}

⚖️ Dose

${medicamento.dose}

💉 Via de administração

${medicamento.via}

📦 Concentração

${medicamento.concentracao}

⚠️ Contraindicações

${medicamento.contraindicacoes}

❗ Efeitos adversos

${medicamento.efeitos}

🔄 Interações medicamentosas

${medicamento.interacoes}

🚨 Toxicidade / intoxicação

${medicamento.toxicidade}

🩺 Antídoto / conduta

${medicamento.antidoto}

📚 Referências

${ medicamento.referencias.length ? medicamento.referencias.join("
") : "A cadastrar" }

`; ficha.scrollIntoView({ behavior: "smooth" }); } function voltarParaBusca() { document .getElementById("ficha") .style.display = "none"; document .getElementById("pesquisa") .focus(); }
